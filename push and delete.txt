//value push and delete(pop)

#include<stdio.h>
int stack[10];
int maxsize = 10;
int top = -1;
int push(int value)
{
    if(top==maxsize-1)
        printf("overflow\n");
    else{

        top = top+1;
        stack[top]=value;
    }
}

int pop()
{
    if(top==-1)
     printf("underflow\n");
    else{
        top = top-1;

    }

}

int main()
{
    push(10);
    push(20);

    for(int i=0;i<=top;i++)
        printf("%d ",stack[i]);
        pop();
           for(int i=0;i<=top;i++)
        printf("%d ",stack[i]);
}
